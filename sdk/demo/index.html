Integration snippet

// 1. Load the SDK in your <head> <link rel="stylesheet" href="rcm-dialer-sdk.css"/> <script src="rcm-dialer-sdk.js"></script> // 2. Mount the widget RCMDialer.mount({ callStartUrl: '/dialer/call/start', callActionUrl: '/dialer/call/action', disconnectUrl: '/dialer/call/end', eventsUrl: '/dialer/events', }); // 3. Make a call RCMDialer.call({ phone: '+919876543210', contactName: 'John Doe' }); // 4. Listen for events RCMDialer.on('call.ended', (ev) => console.log('Call ended:', ev));
